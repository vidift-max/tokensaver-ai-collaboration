const SENSITIVE_KEYS = ['password','token','secret','credential','session_secret','api_key','authorization','cookie'];
function redactSecrets(obj) {
  if (obj == null || typeof obj !== 'object') return obj;
  const out = Array.isArray(obj) ? [] : {};
  for (const [k, v] of Object.entries(obj)) {
    if (SENSITIVE_KEYS.some(s => k.toLowerCase().includes(s))) out[k] = '[REDACTED]';
    else if (v && typeof v === 'object') out[k] = redactSecrets(v);
    else out[k] = v;
  }
  return out;
}
function hasSecretLeak(text) {
  if (typeof text !== 'string') return false;
  return /(Bearer\\s+[A-Za-z0-9._-]{20,}|sk-[a-z0-9]{20,}|session_secret|password\\s*[:=])/i.test(text);

 * WebSocket server MINIMO (RFC6455): handshake + text frames + ping/pong + close.
 * Zero dependencias. Suficiente para eventos em tempo real e testes.

const crypto = require('crypto');
const WS_GUID = '258EAFA5-E914-47DA-95CA-C5AB0DC85B11';

function acceptKey(key) { return crypto.createHash('sha1').update(key + WS_GUID).digest('base64'); }

function encodeFrame(opcode, payload) {
  const data = Buffer.isBuffer(payload) ? payload : Buffer.from(payload);
  const len = data.length; let header;
  if (len < 126) { header = Buffer.from([0x80 | opcode, len]); }
  else if (len < 65536) { header = Buffer.alloc(4); header[0] = 0x80 | opcode; header[1] = 126; header.writeUInt16BE(len, 2); }
  else { header = Buffer.alloc(10); header[0] = 0x80 | opcode; header[1] = 127; header.writeBigUInt64BE(BigInt(len), 2); }
  return Buffer.concat([header, data]);
}
function decodeFrames(buffer) {
  const frames = []; let off = 0;
  while (off + 2 <= buffer.length) {
    const fin = (buffer[off] & 0x80) !== 0, opcode = buffer[off] & 0x0f, masked = (buffer[off + 1] & 0x80) !== 0;
    let len = buffer[off + 1] & 0x7f, pos = off + 2;
    if (len === 126) { if (pos + 2 > buffer.length) break; len = buffer.readUInt16BE(pos); pos += 2; }
    else if (len === 127) { if (pos + 8 > buffer.length) break; len = Number(buffer.readBigUInt64BE(pos)); pos += 8; }
    const maskLen = masked ? 4 : 0;
    if (pos + maskLen + len > buffer.length) break;
    let payload = buffer.slice(pos + maskLen, pos + maskLen + len);
    if (masked) { const mask = buffer.slice(pos, pos + 4); const out = Buffer.alloc(len); for (let i = 0; i < len; i++) out[i] = payload[i] ^ mask[i & 3]; payload = out; }
    frames.push({ fin, opcode, payload }); off = pos + maskLen + len;
  }
  return { frames, rest: buffer.slice(off) };
}

class WsHub {
  constructor() { this.clients = new Set(); this.subscriptions = new Map(); } // roomId -> Set<client>
  add(client) { this.clients.add(client); }
  remove(client) {
    this.clients.delete(client);
    for (const set of this.subscriptions.values()) set.delete(client);
  }
  subscribe(client, roomId) {
    if (!this.subscriptions.has(roomId)) this.subscriptions.set(roomId, new Set());
    this.subscriptions.get(roomId).add(client);
  }
  publish(roomId, event) {
    const set = this.subscriptions.get(roomId); if (!set) return 0;
    const frame = encodeFrame(1, JSON.stringify(event)); let n = 0;
    for (const c of set) { try { c.write(frame); n++; } catch { this.remove(c); } }
    return n;
  }
  stats() { return { clients: this.clients.size, rooms: this.subscriptions.size }; }
}

    const key = req.headers['sec-websocket-key'];

    socket.write('HTTP/1.1 101 Switching Protocols\\r\\nUpgrade: websocket\\r\\nConnection: Upgrade\\r\\n' +

      write: f => { try { socket.write(f); } catch { /* ignore */ } },
      send: obj => client.write(encodeFrame(1, JSON.stringify(obj))),
      close: () => { try { socket.end(); } catch { /* ignore */ } }
    };
    hub.add(client);
    socket.on('data', d => {
      buf = Buffer.concat([buf, d]);
      const { frames, rest } = decodeFrames(buf); buf = rest;
      for (const f of frames) {
        if (f.opcode === 0x8) { hub.remove(client); socket.end(); return; }
        if (f.opcode === 0x9) { client.write(encodeFrame(0xA, f.payload)); continue; } // ping -> pong
        if (f.opcode === 0x1) {
          let msg; try { msg = JSON.parse(f.payload.toString()); } catch { continue; }
          // protocolo minimo do cliente: {op:'subscribe', room_id, participant_id, principal_id}
          if (msg.op === 'subscribe' && msg.room_id) {
            client.roomId = msg.room_id; client.participantId = msg.participant_id || null; client.principalId = msg.principal_id || null;
            hub.subscribe(client, msg.room_id);
            client.send({ type: 'SUBSCRIBED', room_id: msg.room_id, at: new Date().toISOString() });
            if (onConnection) onConnection(client, msg);
          }
        }
      }
    });
    socket.on('close', () => hub.remove(client));
    socket.on('error', () => hub.remove(client));
  });
  return hub;
}

const test = require('node:test');
const assert = require('node:assert');
const crypto = require('crypto');
const { server, store, hub } = require('../backend/src/server');
const { encodeFrame } = require('../backend/src/ws/wsServer');

function wsConnect(port, roomId) {
  return new Promise((resolve, reject) => {
    const key = crypto.randomBytes(16).toString('base64');
    const socket = require('net').connect(port, '127.0.0.1', () => {
      socket.write(`GET /ws HTTP/1.1\\r\\nHost: 127.0.0.1:${port}\\r\\nUpgrade: websocket\\r\\nConnection: Upgrade\\r\\nSec-WebSocket-Key: ${key}\\r\\nSec-WebSocket-Version: 13\\r\\n\\r\\n`);
    });
    let buf = Buffer.alloc(0), upgraded = false;
    const events = [];
    socket.on('data', d => {
      buf = Buffer.concat([buf, d]);
      if (!upgraded) {
        const idx = buf.indexOf('\\r\\n\\r\\n');
        if (idx < 0) return;
        upgraded = true; buf = buf.slice(idx + 4);
        socket.write(encodeFrame(1, JSON.stringify({ op: 'subscribe', room_id: roomId })));
        resolve({ socket, events });
      }
      // parse frames minimas (sem mascara no servidor->cliente)
      while (buf.length >= 2) {
        const len = buf[1] & 0x7f; let pos = 2;
        if (len === 126) { if (buf.length < 4) return; pos = 4; }
        if (buf.length < pos + (len === 126 ? buf.readUInt16BE(2) : len)) return;
        const plen = len === 126 ? buf.readUInt16BE(2) : len;
        try { events.push(JSON.parse(buf.slice(pos, pos + plen).toString())); } catch { /* ignore */ }
        buf = buf.slice(pos + plen);
      }
    });
    socket.on('error', reject);
  });
}
